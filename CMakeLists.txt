cmake_minimum_required(VERSION 3.20)
project(BlackSquelch VERSION 1.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(BS_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Use GCC or Clang; on Windows use MinGW-w64 (MSYS2 UCRT64).")
endif()
add_compile_definitions($<$<NOT:$<PLATFORM_ID:Windows>>:_POSIX_C_SOURCE=200809L>)
if(BS_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(NOT WIN32)
    find_package(Threads REQUIRED)
    set(BS_THREAD_LIBRARY Threads::Threads)
endif()
add_library(bs_options INTERFACE)
target_include_directories(bs_options INTERFACE include)
target_compile_options(bs_options INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wstrict-prototypes)
if(NOT WIN32)
    target_link_libraries(bs_options INTERFACE m)
endif()
add_library(bs_core src/protocol.c src/modem.c src/mfsk.c src/fec.c src/wav.c)
target_link_libraries(bs_core PUBLIC bs_options)
add_library(bs_support src/platform.c src/config.c src/log.c src/commands.c src/files.c src/macros.c)
target_link_libraries(bs_support PUBLIC bs_core)
function(add_audio_library name)
    add_library(${name} STATIC src/audio.c src/audio_queue.c src/audio_driver.c)
    target_link_libraries(${name} PUBLIC bs_options PRIVATE ${BS_THREAD_LIBRARY} ${CMAKE_DL_LIBS})
    target_include_directories(${name} SYSTEM PRIVATE third_party/miniaudio)
endfunction()
add_audio_library(bs_audio)
add_executable(blacksquelch src/main.c src/app.c src/tui.c)
target_link_libraries(blacksquelch PRIVATE bs_support bs_audio)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_core tests/test_core.c)
    target_link_libraries(test_core PRIVATE bs_core)
    add_executable(test_app tests/test_app.c)
    target_link_libraries(test_app PRIVATE bs_support)
    add_executable(test_files tests/test_files.c)
    target_link_libraries(test_files PRIVATE bs_support)
    add_executable(test_mfsk tests/test_mfsk.c)
    target_link_libraries(test_mfsk PRIVATE bs_support)
    add_executable(test_macros tests/test_macros.c)
    target_link_libraries(test_macros PRIVATE bs_support)
    add_library(bs_audio_test STATIC src/audio_driver.c src/audio_queue.c)
    target_compile_definitions(bs_audio_test PUBLIC BS_AUDIO_TEST)
    target_include_directories(bs_audio_test SYSTEM PUBLIC third_party/miniaudio)
    target_link_libraries(bs_audio_test PUBLIC bs_options PRIVATE ${BS_THREAD_LIBRARY} ${CMAKE_DL_LIBS})
    add_executable(test_audio tests/test_audio.c)
    target_include_directories(test_audio PRIVATE src)
    target_link_libraries(test_audio PRIVATE bs_audio_test bs_support)
    add_executable(test_tui tests/test_tui.c src/tui.c)
    target_link_libraries(test_tui PRIVATE bs_support)
    foreach(test core app audio tui files mfsk macros)
        target_compile_options(test_${test} PRIVATE -UNDEBUG)
        add_test(NAME ${test} COMMAND test_${test})
    endforeach()
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(Python3 COMPONENTS Interpreter)
        if(Python3_FOUND)
            add_test(NAME terminal COMMAND ${Python3_EXECUTABLE}
                     ${CMAKE_SOURCE_DIR}/tests/test_terminal.py $<TARGET_FILE:blacksquelch>)
            add_test(NAME file_cli COMMAND ${Python3_EXECUTABLE}
                     ${CMAKE_SOURCE_DIR}/tests/test_file_cli.py $<TARGET_FILE:blacksquelch>)
        endif()
    endif()
endif()
